// --- 4. REAL-TIME DATABASE LISTENERS ---
let unsubActivities, unsubMarket, unsubMyBids;
function subscribeToData() {
    if (unsubActivities) unsubActivities();
    if (unsubMarket) unsubMarket();
    if (unsubMyBids) unsubMyBids();

    unsubMarket = onSnapshot(query(collection(db, `artifacts/${APP_ID}/public/data/listings`), orderBy('createdAt', 'desc')), 
        (snap) => {
            state.listings = snap.docs.map(d => ({id: d.id, ...d.data()}));
            renderMarketplace();
            renderDashboard();
        }, (error) => { showInlineError('mktError', "Marketplace offline."); });

    if (currentRole === 'farmer') {
        unsubActivities = onSnapshot(query(collection(db, `artifacts/${APP_ID}/users/${currentUser.uid}/activities`), orderBy('createdAt', 'desc')), 
            (snap) => {
                state.activities = snap.docs.map(d => ({ id: d.id, ...d.data() }));
                renderActivities();
            }, (error) => { showInlineError('actError', "Could not load activities."); });
    }
    
    if (currentRole === 'buyer') {
        unsubMyBids = onSnapshot(query(collectionGroup(db, 'bids'), where('buyerId', '==', currentUser.uid)), (snap) => {
            state.myBids = snap.docs.map(d => d.data());
            renderDashboard();
        }, (error) => {
            console.error("Error fetching my bids:", error);
            showInlineError('mktError', "Could not load your bid history. Index may be required.");
        });
    }
}

// --- *** MODIFIED BIDDING FUNCTIONS *** ---
async function handleSubmitBid() {
    const bidAmount = parseFloat(document.getElementById('bidAmount').value);
    
    if (!bidAmount || isNaN(bidAmount) || bidAmount <= 0) {
        return showInlineError('bidError', "Please enter a valid bid amount.");
    }
    
    if (bidAmount <= currentHighestBid) {
        return showInlineError('bidError', `Your bid must be higher than the current top bid of ₹${currentHighestBid}`);
    }
    
    if (!currentListingId) {
        return showInlineError('bidError', "Listing ID not found. Please try again.");
    }

    try {
        await addDoc(collection(db, `artifacts/${APP_ID}/public/data/listings/${currentListingId}/bids`), {
            bidAmount: bidAmount,
            buyerEmail: currentUser.email || 'Anonymous',
            buyerId: currentUser.uid,
            createdAt: serverTimestamp()
        });
        document.getElementById('bidModal').classList.add('hidden');
        showMsg("Bid Placed", "Your bid has been successfully placed!");
    } catch (e) {
        showInlineError('bidError', "Error placing bid.");
    }
}

// Helper function to format timestamp
function formatTimestamp(timestamp) {
    if (!timestamp) return 'Just now';
    const date = timestamp.toDate();
    return date.toLocaleString('en-IN', { 
        hour: '2-digit', 
        minute: '2-digit', 
        second: '2-digit',
        day: '2-digit', 
        month: '2-digit' 
    });
}

// NEW: This function opens the modal and sets up a LIVE listener for bids
function openBidModalAndLoadHistory(listingId, startPrice) {
    // Reset inputs and error messages
    document.getElementById('bidAmount').value = '';
    document.getElementById('bidError').classList.add('hidden');
    
    // Show loader
    const loader = document.getElementById('bidListLoader');
    loader.classList.remove('hidden');
    document.getElementById('latestBidsList').innerHTML = '';
    document.getElementById('topBidPrice').textContent = '...';

    // If we already have a listener, stop it
    if (unsubItemBids) unsubItemBids();

    // Listen for real-time bid updates
    const q = query(collection(db, `artifacts/${APP_ID}/public/data/listings/${listingId}/bids`), orderBy('bidAmount', 'desc'));
    unsubItemBids = onSnapshot(q, (snap) => {
        const item = state.listings.find(i => i.id === listingId);
        const bids = snap.docs.map(d => ({...d.data(), id: d.id}));
        renderBidModalUI(bids, item.price);
    }, (error) => {
        console.error("Error listening to bids:", error);
        loader.classList.add('hidden');
        document.getElementById('latestBidsList').innerHTML = '<p style="color:red;">Could not load bids.</p>';
    });
}

// NEW: This function renders the UI *inside* the bid modal
function renderBidModalUI(bids, startPrice) {
    document.getElementById('bidListLoader').classList.add('hidden');
    const listEl = document.getElementById('latestBidsList');
    const topPriceEl = document.getElementById('topBidPrice');
    const bidInput = document.getElementById('bidAmount');

    // Find the highest bid
    let maxBid = parseFloat(startPrice);
    if (bids.length > 0) {
        maxBid = Math.max(maxBid, ...bids.map(b => b.bidAmount));
    }
    currentHighestBid = maxBid;
    
    // Update Top Price and Input Placeholder
    topPriceEl.textContent = `₹${currentHighestBid}`;
    bidInput.placeholder = `> ₹${currentHighestBid}`;

    // Render Bid List
    if (bids.length === 0) {
        listEl.innerHTML = `<p style="text-align:center;color:var(--muted);padding:20px 0;">No bids yet. Be the first!</p>`;
    } else {
        listEl.innerHTML = bids.map(bid => `
            <div class="bid-item">
                <div class="bid-item-details">
                    <strong>${bid.buyerEmail.split('@')[0]}</strong>
                    <div class="bid-item-time">${formatTimestamp(bid.createdAt)}</div>
                </div>
                <div class="bid-item-price">₹${bid.bidAmount}</div>
            </div>
        `).join('');
    }
}
